## ✅ Code (Python)

```python
class Solution:
    def hasVowel(self, word: str) -> bool:
        vowels = {'a', 'e', 'i', 'o', 'u'}
        for vowel in vowels:
            if vowel in word.lower():
                return True

        return False

    def hasConsonant(self, word: str) -> bool:
        vowels = {'a', 'e', 'i', 'o', 'u'}
        for letter in word.lower():
            if letter not in vowels and letter.isalpha():
                return True

        return False

    def isValid(self, word: str) -> bool:
        if len(word) < 3:
            return False

        for letter in word.lower():
            if letter.isalnum():
                continue
            else:
                return False

        return self.hasVowel(word) and self.hasConsonant(word)
```
